# 1.blender处理
在blender中调整stl位置确定坐标，然后用脚本批量导出

```python
import bpy, os

out_dir = r"E:/Document/RoboMaster/Wheelleg/stl"   # 按需改
names = ["body","L1","L2","L3","L4","L5","L6","LW",
         "R1","R2","R3","R4","R5","R6","RW"]

for name in names:
    ob = bpy.data.objects.get(name)
    if ob is None:
        print("缺少对象:", name); continue
    bpy.ops.object.select_all(action='DESELECT')
    ob.select_set(True)                        # 每次只选当前这一个零件
    path = os.path.join(out_dir, name + ".stl")

    try:                                       # Blender >= 3.1 (wm.stl_export)
        bpy.ops.wm.stl_export(
            filepath=path, check_existing=False,
            export_selected_objects=True,      # 关键: 只导出选中的这一件
            global_scale=1.0)
    except TypeError:
        try:                                   # 参数名不同的版本, 换 use_selection
            bpy.ops.wm.stl_export(filepath=path, check_existing=False,
                                  use_selection=True)
        except Exception as e:
            print(name, "导出失败:", e); continue
    except AttributeError:                     # Blender <= 3.0 旧接口
        bpy.ops.export_mesh.stl(filepath=path, use_selection=True)
    print("已导出", name)

```
